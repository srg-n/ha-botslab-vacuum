#
# pytest-homeassistant-custom-component pins a specific Home Assistant release,
# so both are pinned together to keep the tested version unambiguous:
#   0.13.371 -> Home Assistant 2026.10.0  (requires Python 3.14)
pytest-homeassistant-custom-component==0.13.371
pytest>=8.0
pytest-asyncio>=0.24

# homeassistant.components.camera imports TurboJPEG at module scope, with no
# guard, so the camera platform cannot be imported without it. Home Assistant
# ships this as part of requirements_all, which a pip install of the package
# does not pull in. 2.5.0 is pinned because it is a pure Python wheel; the
# versions Home Assistant itself pins (1.8.x) are sdists that compile.
# numpy, which PyTurboJPEG needs at import, comes from the pinned
# pytest-homeassistant-custom-component.
PyTurboJPEG==2.5.0
